# SRB NMS希望与我们重新组建联邦
country_event = {
	id = nms.111
	title = nms.111.t
	desc = nms.111.d
	major = no
	fire_only_once = no
	is_triggered_only = yes

	option = {
		name = nms.111.1
		ai_chance = {
			base = 10
			modifier = {
				add = 1000
				strength_ratio = {
					tag = NMS
					ratio < 0.5
				}
			}
		}
		if = {
			limit = {
				NMS = {
					is_ai = yes
				}
			}
			custom_effect_tooltip = NMS_SRB_transfer_to_NMS_tt
			hidden_effect = {
				NMS = {
					change_tag_from = ROOT
				}
			}
		}
		else = {
			custom_effect_tooltip = NMS_SRB_game_over_warning_tt
		}
		hidden_effect = {
			NMS = {
				country_event = nms.114
				set_cosmetic_tag = NMS_SRB
				# news
				news_event = {
					id = nms.124
					days = 5
				}
			}
			# 将领移交
			SRB = {
				every_character = {
					set_nationality = NMS
					clr_character_flag = nationality_SRB
            		set_character_flag = nationality_NMS
				}
			}
			# 造核
			SRB = {
				every_core_state = {
					add_core_of = NMS
				}
			}
			# 吞并
			NMS = {
				annex_country = {
					target = SRB
					transfer_troops = yes
				}
			}
		}
	}

	option = {
		name = nms.111.2
		ai_chance = {
			base = 1
			modifier = {
				add = -10
				strength_ratio = {
					tag = NMS
					ratio < 0.5
				}
			}
			modifier = {
				add = 100
				strength_ratio = {
					tag = NMS
					ratio > 1
				}
			}
		}
		hidden_effect = {
			NMS = {
				country_event = nms.113
			}
		}
	}
}
