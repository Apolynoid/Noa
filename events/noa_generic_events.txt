country_event = {
	id = noa_gen.112
	hidden = yes
	is_triggered_only = no

	trigger = {
		has_country_flag = NOA_faction_collapse_flag
	}

	immediate = {
		clr_country_flag = NOA_faction_collapse_flag
	}

	option = {
		# search for negotiate target
		get_sorted_scored_countries_temp = {
			scorer = most_powerful_country
			array = power_list
		}
		for_loop_effect = {
			end = power_list^num
			value = s
			break = t_break
			var:power_list^s = {
				if = {
					limit = {
						has_war_with = ROOT
						is_subject = no
						OR = {
							is_in_faction = no
							is_faction_leader = yes
						}
					}
					save_event_target_as = NOA_winner_target
					ROOT = {
						country_event = noa_gen.108
					}
				}
			}
		}
	}
}
