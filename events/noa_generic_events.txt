	id = noa_gen.103
	title = noa_gen.103.t
	desc = noa_gen.103.d
	is_triggered_only = yes

	# 我们将齐心协力
	option = {
		name = noa_gen.103.1
		FROM = {
			add_to_faction = ROOT
		}
		hidden_effect = {
			FROM = {
				country_event = noa_gen.104
			}
		}
		ai_chance = {
			base = 1
			# don't join faction when you already has one
			modifier = {
				factor = 0
				OR = {
					is_subject = yes
					is_in_faction = yes
				}
			}
		}
	}

	# 我们不能冒这种风险
	option = {
		name = noa_gen.103.2
		hidden_effect = {
			FROM = {
				country_event = noa_gen.105
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					is_subject = yes
					is_in_faction = yes
				}
			}
		}
	}
}
