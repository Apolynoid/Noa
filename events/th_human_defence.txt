country_event = { #保护人类
	id = hud.3
	title = hud.3.t
	desc = hud.3.d

	fire_only_once = no
	is_triggered_only = no
	mean_time_to_happen = {
		days = 3
	}

	trigger = {
		or = {
			HUM = { is_subject = yes }
			not = {
				is_neighbor_of = HUM
				any_country = {
					and = {
						is_subject_of = ROOT
						is_neighbor_of = HUM
					}
				}
				
			}
		}
		has_global_flag = main_scenario
		has_guaranteed = HUM
	}

	option = {
		name = hud.3.1
		diplomatic_relation = {
		    country = HUM
		    relation = guarantee
		    active = no
		}
	}
}

country_event = { #保护人类(阵营结束)
	id = hud.6
	title = hud.6.t
	desc = hud.6.d

	trigger = {
		has_country_flag = HUM_defence
		has_global_flag = main_scenario
		is_in_faction = yes
		not = {
			has_war_together_with = HUM
			has_country_flag = HUM_defence_not_dismantle
		}
	}
	mean_time_to_happen = {
		days = 2
	}

	option = {
		name = hud.6.1
		dismantle_faction = yes
		every_country = {
			limit = {
				has_country_flag = HUM_defence_not_alliance
			}
			add_ai_strategy = {
				type = alliance
				id = ROOT
				value = 10000
			}
			ROOT = {
				add_ai_strategy = {
					type = alliance
					id = PREV
					value = 10000
				}
			}
			clr_country_flag = HUM_defence_not_alliance
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = hud.6.2
		set_country_flag = HUM_defence_not_dismantle
		ai_chance = {
			base = 0
		}
	}
}
