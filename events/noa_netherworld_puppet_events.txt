add_namespace = netpup

# YYK 231
country_event = {
	id = netpup.101
	title = netpup.101.t
	desc = netpup.101.d

	fire_only_once = no
	is_triggered_only = no

	trigger = {
		has_war = no
		NOT = {
			has_country_flag = YYK_puppet_event_flag
		}
		NOT = {
			country_exists = YYK
		}
		controls_state = 231
	}

	mean_time_to_happen = {
		days = 5
	}

	immediate = {
		hidden_effect = {
			set_country_flag = YYK_puppet_event_flag
		}
	}

	# 在首都释放一个傀儡国
	option = {
		name = netpup.101.1
		YYK = {
			transfer_state = 231
		}
		puppet = YYK
	}

	# 释放一个傀儡国管理所有有关地区
	option = {
		name = netpup.101.2
		trigger = {
			any_owned_state = {
				is_core_of = YYK
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = YYK
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
			transfer_state_to = YYK
		}
		puppet = YYK
	}

	# 维持军事占领
	option = {
		name = netpup.101.3
		add_stability = -0.05
		ai_chance = {
			base = 0
		}
	}
}
