# SCA 反对者的破坏行动
country_event = {
	id = cir.111
	title = cir.111.t
	desc = cir.111.d
	is_triggered_only = yes
	option = {
		name = cir.111.1
		if = {
			limit = {
				SCA = {
					NOT = {
						has_country_flag = SCA_rebellion_system_flag
					}
					NOT = {
						has_completed_focus = SCA_zhen_ya_fan_dui_zhe
					}
				}
			}
			# less damage
			SCA = {
				random_core_state = {
					limit = {
						is_fully_controlled_by = SCA
						non_damaged_building_level = {
							building = industrial_complex
							level > 0
						}
					}
					damage_building = {
						type = industrial_complex
						damage = 1
					}
				}
				random_core_state = {
					limit = {
						is_fully_controlled_by = SCA
						non_damaged_building_level = {
							building = industrial_complex
							level > 0
						}
					}
					damage_building = {
						type = industrial_complex
						damage = 1
					}
				}
			}
		}
		else = {
			# more damage with militias
			hidden_effect = {
				CIR = {
					delete_unit_template_and_units = {
						division_template = "妖精起义者"
						disband = yes
					}
					division_template = {
						name = "妖精起义者"
						is_locked = no
						priority = 0
						regiments = {
							ma_infantry = { x = 0 y = 0 }
							ma_light_infantry = { x = 0 y = 1 }
							ma_light_infantry = { x = 0 y = 2 }
						}
					}
				}
			}
			SCA = {
				for_loop_effect = {
					end = 3
					random_core_state = {
						limit = {
							is_fully_controlled_by = SCA
							non_damaged_building_level = {
								building = industrial_complex
								level > 0
							}
						}
						damage_building = {
							type = industrial_complex
							damage = 1
						}
						custom_effect_tooltip = CIR_CIR_create_fairy_rebellion_tt
						hidden_effect = {
							create_unit = {
								division = "name = \"妖精起义者\" division_template = \"妖精起义者\" start_manpower_factor = 1"
								owner = CIR
								allow_spawning_on_enemy_provs = yes
							}
						}
					}
				}
			}
		}
		if = {
			limit = {
				has_country_flag = SCA_rebellion_system_flag
			}
			SCA_fairy_rebellion_minus_10 = yes
			SCA_fairy_rebellion_minus_10 = yes
		}
	}
}
