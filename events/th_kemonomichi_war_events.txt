country_event = {
	id = kem_war.105
	title = kem_war.105.t
	desc = kem_war.105.d
	fire_only_once = yes
	is_triggered_only = no

	trigger = {
		# generic triggers
		has_global_flag = MSD_kemonomichi_war_begun_flag
		NOT = {
			has_global_flag = MSD_kemonomichi_war_ended_flag
		}
		NOT = {
			has_global_flag = MSD_kemonomichi_war_fate_chosen_flag
		}
		# check
		original_tag = MSD
		OR = {
			206 = {
				is_fully_controlled_by = MSD
			}
			MRS = {
				surrender_progress > 0.2
			}
		}
	}

	mean_time_to_happen = {
		days = 5
	}

	immediate = {
		hidden_effect = {
			set_global_flag = MSD_kemonomichi_war_fate_chosen_flag
			set_global_flag = MSD_kemonomichi_war_ended_flag
			MSD = {
				white_peace = MRS
				transfer_state = 206
				set_country_flag = MSD_victory_in_kemonomichi_war_flag
			}
			MRS = {
				country_event = kem_war.104
			}
			news_event = {
				id = kem_war.109
				days = 7
			}
		}
	}

	option = {
		name = kem_war.105.1
	}
}
