# 河童内战继续
country_event = {
	id = kp.108
	title = kp.108.t
	desc = kp.108.d

	is_triggered_only = no

	trigger = {
		OR = {
			original_tag = NTR
			original_tag = KAP
		}
		OR = {
			has_war_with = NTR
			has_war_with = KAP
		}
		has_country_flag = KP_can_act_in_reunification_flag
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = KP_can_act_in_reunification_flag
			if = {
				limit = {
					original_tag = NTR
				}
				every_owned_state = {
					limit = {
						check_variable = {
							FROM.KP_paramilitary_KAP > 0.3
						}
					}
					damage_units = {
						state = THIS
						damage = 0.05
						limit = {
							tag = NTR
						}
						ratio = yes
						army = yes
						navy = no
					}
				}
			} else_if = {
				limit = {
					original_tag = KAP
				}
				every_owned_state = {
					limit = {
						check_variable = {
							FROM.KP_paramilitary_NTR > 0.3
						}
					}
					damage_units = {
						state = THIS
						damage = 0.05
						limit = {
							tag = KAP
						}
						ratio = yes
						army = yes
						navy = no
					}
				}
			}
		}
	}

	option = {
		name = kp.108.1
		custom_effect_tooltip = KP_paramilitary_sabotage_tt
	}
}

# 河童再统一运动结束
country_event = {
	id = kp.109
	title = kp.109.t
	desc = kp.109.d

	is_triggered_only = no

	trigger = {
		OR = {
			original_tag = NTR
			original_tag = KAP
			original_tag = YAW
		}
		has_country_flag = KP_reunification_process_flag
		NOT = {
			count_triggers = {
				amount = 2
				KAP = {
					NOA_is_independent = yes
				}
				NTR = {
					NOA_is_independent = yes
				}
				YAW = {
					NOA_is_independent = yes
					has_country_flag = KP_reunification_process_flag
				}
			}
		}
	}

	immediate = {
		hidden_effect = {
			every_country = {
				limit = {
					has_country_flag = KP_reunification_process_flag
				}
				clr_country_flag = KP_reunification_process_flag
			}
		}
	}

	option = {
		name = kp.109.1
	}
}
