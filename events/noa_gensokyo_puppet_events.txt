# SCA 122
country_event = {
	id = genpup.101
	title = genpup.101.t
	desc = genpup.101.d

	fire_only_once = no
	is_triggered_only = no

	trigger = {
		has_war = no
		NOT = {
			has_country_flag = SCA_puppet_event_flag
		}
		NOT = {
			country_exists = SCA
		}
		controls_state = 122
	}

	mean_time_to_happen = {
		days = 5
	}

	immediate = {
		hidden_effect = {
			set_country_flag = SCA_puppet_event_flag
		}
	}

	# 在首都释放一个傀儡国
	option = {
		name = genpup.101.1
		SCA = {
			transfer_state = 122
		}
		puppet = SCA
	}

	# 释放一个傀儡国管理所有有关地区
	option = {
		name = genpup.101.2
		trigger = {
			any_owned_state = {
				is_core_of = SCA
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = SCA
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
			transfer_state_to = SCA
		}
		puppet = SCA
	}

	# 维持军事占领
	option = {
		name = genpup.101.3
		add_stability = -0.05
		ai_chance = {
			base = 0
		}
	}
}
