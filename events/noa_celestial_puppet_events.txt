add_namespace = celpup

# TES 49
country_event = {
	id = celpup.101
	title = celpup.101.t
	desc = celpup.101.d

	fire_only_once = no
	is_triggered_only = no

	trigger = {
		has_war = no
		NOT = {
			has_country_flag = TES_puppet_event_flag
		}
		NOT = {
			country_exists = TES
		}
		controls_state = 49
	}

	mean_time_to_happen = {
		days = 5
	}

	immediate = {
		hidden_effect = {
			set_country_flag = TES_puppet_event_flag
		}
	}

	# 在首都释放一个傀儡国
	option = {
		name = celpup.101.1
		TES = {
			transfer_state = 49
		}
		puppet = TES
	}

	# 释放一个傀儡国管理所有有关地区
	option = {
		name = celpup.101.2
		trigger = {
			any_owned_state = {
				is_core_of = TES
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = TES
				NOT = {
					is_core_of = ROOT
				}
				NOT = {
					is_claimed_by = ROOT
				}
			}
			transfer_state_to = TES
		}
		puppet = TES
	}

	# 维持军事占领
	option = {
		name = celpup.101.3
		add_stability = -0.05
		ai_chance = {
			base = 0
		}
	}
}
