add_namespace = tegcw

# 天狗内战开始
news_event = {
	id = tegcw.101
	title = tegcw.101.t
	desc = tegcw.101.d

	fire_only_once = no
	is_triggered_only = no
	major = yes

	trigger = {
		TEG = {
			exists = yes
			is_subject = no
		}
		OTG = {
			exists = yes
			is_subject = no
		}
		TEG = {
			has_war_with = OTG
		}
		NOT = {
			has_global_flag = TEG_civil_war_flag
		}
	}

	immediate = {
		hidden_effect = {
			set_global_flag = TEG_civil_war_flag
			TEG_war_progress_add_10 = yes
		}
	}

	mean_time_to_happen = {
		days = 5
	}

	# 天狗共和国
	option = {
		name = tegcw.101.1
		trigger = {
			original_tag = TEG
		}
	}

	# 天魔帝国
	option = {
		name = tegcw.101.2
		trigger = {
			original_tag = OTG
		}
	}

	# 其他
	option = {
		name = tegcw.101.3
		trigger = {
			NOT = {
				OR = {
					original_tag = TEG
					original_tag = OTG
				}
			}
		}
	}
}

		set_global_flag = TEG_get_7539_flag
