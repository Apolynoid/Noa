# HUM SRB请求合并
country_event = {
	id = srb.102
	title = srb.102.t
	desc = srb.102.d
	major = no
	fire_only_once = no
	is_triggered_only = yes

	option = {
		name = srb.102.1
		ai_chance = {
			base = 10
			# power
			modifier = {
				add = -1
				strength_ratio = {
					tag = SRB
					ratio > 2
				}
			}
			modifier = {
				add = -9
				strength_ratio = {
					tag = SRB
					ratio > 1
				}
			}
			modifier = {
				add = 5
				strength_ratio = {
					tag = SRB
					ratio < 0.5
				}
			}
			# opinion
			modifier = {
				factor = 2
				is_in_faction_with = SRB
			}
			modifier = {
				factor = 2
				has_opinion = {
					target = THIS
					value > 30
				}
			}
			modifier = {
				factor = 2
				has_opinion = {
					target = THIS
					value > 70
				}
			}
		}
		if = {
			limit = {
				SRB = {
					is_ai = yes
				}
			}
			custom_effect_tooltip = SRB_HUM_transfer_to_SRB_tt
			hidden_effect = {
				SRB = {
					change_tag_from = ROOT
				}
			}
		}
		else = {
			custom_effect_tooltip = SRB_HUM_game_over_warning_tt
		}
		hidden_effect = {
			SRB = {
				country_event = srb.104
				news_event = {
					id = srb.105
					days = 5
				}
			}
			# 将领移交
			HUM = {
				every_character = {
					set_nationality = SRB
					clr_character_flag = nationality_HUM
            		set_character_flag = nationality_SRB
				}
			}
			# 造核
			HUM = {
				every_core_state = {
					add_core_of = SRB
				}
			}
			# 吞并
			SRB = {
				annex_country = {
					target = HUM
					transfer_troops = yes
				}
			}
		}
	}

	option = {
		name = srb.102.2
		ai_chance = {
			base = 1
			# if guaranteed, much less likely to be annexed
			modifier = {
				add = 100
				SRB = {
					any_other_country = {
						has_guaranteed = HUM
					}
				}
			}
			# faction can stop annexation
			modifier = {
				factor = 5
				is_in_faction = yes
				NOT = {
					is_in_faction_with = SRB
				}
			}
		}
		hidden_effect = {
			SRB = {
				country_event = srb.103
			}
		}
	}
}
