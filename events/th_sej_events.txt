# 来自SEJ.GetFactionName的支援
country_event = {
	id = sej.137
	title = sej.137.t
	desc = sej.137.d
	is_triggered_only = yes

	option = {
		name = sej.137.1
		if = {
			limit = {
				check_variable = {
					SEJ.support_science_weapon > 0
				}
			}
			add_equipment_to_stockpile = {
				type = gun_a
				amount = var:SEJ.support_science_weapon
			}
		}
		if = {
			limit = {
				check_variable = {
					SEJ.support_magic_weapon > 0
				}
			}
			add_equipment_to_stockpile = {
				type = magic_core_a
				amount = var:SEJ.support_magic_weapon
			}
		}
		add_manpower = var:SEJ.support_manpower
		army_experience = var:SEJ.support_xp
		ai_chance = {
			base = 1
		}
	}
}
