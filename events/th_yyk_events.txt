# 冥界内战结束
country_event = {
	id = yyk.106
	title = yyk.106.t
	desc = yyk.106.d
	is_triggered_only = yes

	option = {
		name = yyk.106.1
		# disband militia
		custom_effect_tooltip = disband_militia_tt
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = YYK_KNA_flag
				}
				set_division_template_lock = {
					division_template = "革命民兵"
					is_locked = no
				}
				delete_unit_template_and_units = {
					division_template = "革命民兵"
					disband = no
				}
			}
			else = {
				set_division_template_lock = {
					division_template = "效忠派民兵"
					is_locked = no
				}
				delete_unit_template_and_units = {
					division_template = "效忠派民兵"
					disband = no
				}
			}
		}
		# add state modifier & count
		hidden_effect = {
			set_variable = {
				YYK_opposition_num = 0
			}
		}
		if = {
			limit = {
				has_country_flag = YYK_KNA_flag
			}
			every_owned_state = {
				limit = {
					has_state_flag = YYK_support_YKO_in_civil_war_flag
				}
				add_dynamic_modifier = {
					modifier = YYK_fan_dui_zhe_di_kang_state_idea
				}
				hidden_effect = {
					ROOT = {
						add_to_variable = {
							YYK_opposition_num = 1
						}
					}
				}
			}
		}
		else = {
			every_owned_state = {
				limit = {
					has_state_flag = YYK_support_KNA_in_civil_war_flag
				}
				add_dynamic_modifier = {
					modifier = YYK_fan_dui_zhe_di_kang_state_idea
				}
				hidden_effect = {
					ROOT = {
						add_to_variable = {
							YYK_opposition_num = 1
						}
					}
				}
			}
		}
		# calculation
		hidden_effect = {
			YYK_opposition_resistance_calculation = yes
		}
		# add ideas
		add_ideas = {
			YYK_nei_zhan_shang_hen_idea
			YYK_nei_zhan_hou_de_sheng_chan_wen_ti_idea
		}
		if = {
			limit = {
				has_country_flag = YYK_YKO_flag
			}
			add_ideas = YKO_si_ji_de_zhi_cai_idea
		}
		add_dynamic_modifier = {
			modifier = YYK_fan_dui_zhe_di_kang_idea
		}
	}
}
