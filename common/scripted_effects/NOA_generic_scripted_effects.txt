NOA_increase_conscription_to_lv3 = {
    if = {
        limit = {
            OR = {
                has_idea = Fairy
            }
        }
        if = {
            limit = {
                OR = {
                    has_idea = manpowerlv1
                    has_idea = type1_manpowerlv2
                }
            }
            add_ideas = type1_manpowerlv3
        }
    } else_if = {
        limit = {

                has_idea = Magician
                has_idea = Human
            }
        }

                OR = {

                }

        }
    } else_if = {
        limit = {
            OR = {
                has_idea = Pri_youkai

            }
        }
        if = {
            limit = {
                OR = {
                    has_idea = manpowerlv1
                    has_idea = type3_manpowerlv2
                }
            }
            add_ideas = type3_manpowerlv3
        }
    } else_if = {
        limit = {
            OR = {
                has_idea = God

                has_idea = Beast
            }
        }
        if = {
            limit = {
                OR = {
