MRS_NAR_land_split = {
    every_state = {
        limit = {
            is_owned_by = MRS
            OR = {
                is_core_of = MRS
                is_claimed_by = MRS
            }
        }
        if = {
            limit = {
                is_claimed_by = NAR
            }
            add_core_of = NAR
            remove_core_of = MRS
            add_claim_by = MRS
            teleport_armies = {
                limit = {
                    tag = MRS
                }
            }
            transfer_state_to = NAR
        }
        else = {
            add_claim_by = NAR
        }
    }
    every_state = {
        limit = {
            is_owned_by = ALI
            OR = {
                is_core_of = ALI
                is_claimed_by = ALI
            }
        }
        add_claim_by = NAR
    }
}

        MRS_remove_all_templates = yes

    }

        value = 0.3

    }

    NAR = {
        inherit_technology = MRS
    }
    # 领导人变动
    NAR_narumi = {
        set_nationality = NAR

        clr_character_flag = nationality_MRS

        set_nationality = NAR

        promote_character = {
            character = NAR_narumi
