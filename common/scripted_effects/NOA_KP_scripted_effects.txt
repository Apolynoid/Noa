KP_calc_legitimacy_reward = {
    set_variable = {
        global.KP_num_country_available = 0
    }
    every_country = {
        limit = {
            KP_has_reunification_process = yes
        }
        add_to_variable = {
            global.KP_num_country_available = 1
        }
    }
    set_variable = {
        global.KP_legitimacy_sum = 0
    }
    every_country = {
        limit = {
            KP_has_reunification_process = yes
        }
        add_to_variable = {
            global.KP_legitimacy_sum = KP_legitimacy
        }
    }
    set_variable = {
        KP_legitimacy_reward = KP_legitimacy
    }
     multiply_variable = {
        KP_legitimacy_reward = global.KP_num_country_available
    }
    divide_variable = {
        KP_legitimacy_reward = global.KP_legitimacy_sum
    }
    subtract_from_variable = {
        KP_legitimacy_reward = 1
    }
    multiply_variable = {
        KP_legitimacy_reward = 4 # competition factor
    }
    add_to_variable = {
        KP_legitimacy_reward = 1
    }
    multiply_variable = {
        KP_legitimacy_reward = 0.07
    }
}

KP_paramilitary_conflict = {
    set_temp_variable = {
        current_NTR_power = { value = FROM.KP_paramilitary_NTR multiply = FROM.KP_paramilitary_NTR }
    }
    set_temp_variable = {
        currrnt_KAP_power = { value = FROM.KP_paramilitary_KAP multiply = FROM.KP_paramilitary_KAP }
    }
    if = {
        limit = {
            check_variable = {
                FROM.KP_paramilitary_NTR > FROM.KP_paramilitary_KAP
            }
        }
        multiply_variable = {
            FROM.KP_paramilitary_KAP = 0.4
        }
        set_temp_variable = {
            final_KAP_power = { value = FROM.KP_paramilitary_KAP multiply = FROM.KP_paramilitary_KAP }
        }
        subtract_from_temp_variable = {
            current_NTR_power = { value = currrnt_KAP_power subtract = final_KAP_power }
        }
        set_variable = {
            FROM.KP_paramilitary_NTR = { value = current_NTR_power root = 2 }
        }
    } else = {
        multiply_variable = {
            FROM.KP_paramilitary_NTR = 0.4
        }
        set_temp_variable = {
            final_NTR_power = { value = FROM.KP_paramilitary_NTR multiply = FROM.KP_paramilitary_NTR }
        }
        subtract_from_temp_variable = {
            current_KAP_power = { value = current_NTR_power subtract = final_NTR_power }
        }
        set_variable = {
            FROM.KP_paramilitary_KAP = { value = current_KAP_power root = 2 }
        }
    }
}
