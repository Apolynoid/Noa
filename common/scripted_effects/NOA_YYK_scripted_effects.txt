YYK_random_state_seize_choose = {
    set_variable = {
        YYK_AI_target_state = 231
    }
    random_owned_state = {
        limit = {
            # not fully controlled
            YYK_state_totally_controlled = no
            # neighboring enemy core
            YYK_is_neighboring_enemy_core_state = yes
            # not being seized
            NOT = {
                has_state_flag = YYK_being_influenced_flag
            }
            # do not seize capital neighbor -- focus will do
            NOT = {
                is_in_array = {
                    array = hakugyokurou_states
                    value = THIS
                }
            }
        }
        set_variable = {
            ROOT.YYK_AI_target_state = THIS
        }
    }
    if = {
        limit = {
            check_variable = {
                YYK_AI_target_state = 231
            }
        }
        # no valid state
        log = "YYK_AI_state: No Valid State!"
        set_country_flag = YYK_no_valid_state_flag
    }
    else = {
        # some valid state
        log = "YYK_AI_state: trying to activate [?YYK_AI_target_state.GetName] mission"
        var:YYK_AI_target_state = {
            set_state_flag = YYK_being_influenced_flag
        }
        activate_mission = YYK_di_fang_xing_dong_zheng_qu__YYK_AI_target_state__di_qu_zhi_chi
    }
}

YYK_KNA_further_action = {
    if = {
        limit = {
            YYK_can_KNA_seek_army_support = no
            YYK_can_seize_weapon = no
        }
        # no further action
        set_country_flag = YYK_no_valid_action_flag
    }
    else_if = {
        limit = {
            YYK_can_KNA_seek_army_support = no
        }
        # seize weapon now
        activate_mission = YYK_di_fang_xing_dong_duo_qu_wu_qi
    }
    else_if = {
        limit = {
            YYK_can_seize_weapon = no
        }
        # seek army support now
        activate_mission = YYK_di_fang_xing_dong_xun_qiu_jun_dui_zhi_chi
    }
    else = {
        # calc weight
        YYK_weapon_action_weight_calc = yes
        set_variable = {
            YYK_AI_army_weight = 50
        }
        if = {
            limit = {
                check_variable = {
                    KNA_army_support > 92
                }
            }
            multiply_variable = {
                YYK_AI_army_weight = 0.1
            }
        }
        # random choose between two options
        random_list = {
            YYK_AI_weapon_weight = {
                activate_mission = YYK_di_fang_xing_dong_duo_qu_wu_qi
            }
            YYK_AI_army_weight = {
                activate_mission = YYK_di_fang_xing_dong_xun_qiu_jun_dui_zhi_chi
            }
        }
    }
}
