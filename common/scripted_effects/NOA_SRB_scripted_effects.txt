SRB_cancel_investment = {
    custom_effect_tooltip = SRB_cancel_investmeny_tt
    hidden_effect = {
        set_country_flag = SRB_cancelled_investment_flag
        set_variable = {
            SRB_influence_value = 0
        }
        SRB_clear_investment = yes
    }
}

    hidden_effect = {
        add_to_variable = {
            SRB_influence_value = SRB_influence_plus
        }

            limit = {

                    SRB_influence_value > 100
                }

            save_event_target_as = SRB_influencial_overlord

            }
