TKM_do_calc_feature = {
    # industrial
    if = {
        limit = {
            check_variable = {
                TKM_industrial_feature_level = 0
            }
        }
        TKM_set_industrial_feature_0 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_industrial_feature_level = 1
            }
        }
        TKM_set_industrial_feature_1 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_industrial_feature_level = 2
            }
        }
        TKM_set_industrial_feature_2 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_industrial_feature_level = 3
            }
        }
        TKM_set_industrial_feature_3 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_industrial_feature_level = 4
            }
        }
        TKM_set_industrial_feature_4 = yes
    } else = {
        log = "TKM Unknown industrial Level"
    }
    # military
    if = {
        limit = {
            check_variable = {
                TKM_military_feature_level = 0
            }
        }
        TKM_set_military_feature_0 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_military_feature_level = 1
            }
        }
        TKM_set_military_feature_1 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_military_feature_level = 2
            }
        }
        TKM_set_military_feature_2 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_military_feature_level = 3
            }
        }
        TKM_set_military_feature_3 = yes
    } else_if = {
        limit = {
            check_variable = {
                TKM_military_feature_level = 4
            }
        }
        TKM_set_military_feature_4 = yes
    } else = {
        log = "TKM Unknown Military Level"
    }
    # calc cost
    set_variable = {
        TKM_monthly_sp_bead_cost = 0
    }
    set_temp_variable = {
        t = TKM_military_feature_level
    }
    multiply_temp_variable = {
        t = TKM_military_feature_level
    }
    add_to_variable = {
        TKM_monthly_sp_bead_cost = t
    }
    set_temp_variable = {
        t = TKM_industrial_feature_level
    }
    multiply_temp_variable = {
        t = TKM_industrial_feature_level
    }
    add_to_variable = {
        TKM_monthly_sp_bead_cost = t
    }
    # AI calc
    set_variable = {
        TKM_reserved_sp_bead = fuel_k
    }
    multiply_variable = {
        TKM_reserved_sp_bead = 1000
    }
    divide_variable = {
        TKM_reserved_sp_bead = TKM_energy_per_sp_bead
    }
    add_to_variable = {
        TKM_reserved_sp_bead = TKM_sp_bead_num
    }
    # now decide the level
    set_temp_variable = {
        t_low_level = 0
    }
    set_temp_variable = {
        t_high_level = 0
    }
    if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 104
            }
        }
        set_temp_variable = {
            t_high_level = 4
        }
        set_temp_variable = {
            t_low_level = 4
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 72
            }
        }
        set_temp_variable = {
            t_high_level = 4
        }
        set_temp_variable = {
            t_low_level = 3
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 47
            }
        }
        set_temp_variable = {
            t_high_level = 3
        }
        set_temp_variable = {
            t_low_level = 3
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 29
            }
        }
        set_temp_variable = {
            t_high_level = 3
        }
        set_temp_variable = {
            t_low_level = 2
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 16
            }
        }
        set_temp_variable = {
            t_high_level = 2
        }
        set_temp_variable = {
            t_low_level = 2
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 8
            }
        }
        set_temp_variable = {
            t_high_level = 2
        }
        set_temp_variable = {
            t_low_level = 1
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 3
            }
        }
        set_temp_variable = {
            t_high_level = 1
        }
        set_temp_variable = {
            t_low_level = 1
        }
    } else_if = {
        limit = {
            check_variable = {
                TKM_reserved_sp_bead > 1
            }
        }
        set_temp_variable = {
            t_high_level = 1
        }
        set_temp_variable = {
            t_low_level = 0
        }
    }
    if = {
        limit = {
            OR = {
                has_war = yes
                date > 148.1.1
            }
            OR = {
                SCA = {
                    exists = yes
                    is_subject = no
                }
                NMS = {
                    exists = yes
                    is_subject = no
                }
            }
        }
        set_variable = {
            TKM_AI_military_level = t_high_level
        }
        set_variable = {
            TKM_AI_industrial_level = t_low_level
        }
    } else = {
        set_variable = {
            TKM_AI_military_level = t_low_level
        }
        set_variable = {
            TKM_AI_industrial_level = t_high_level
        }
    }
}
