faction_allies_core_and_claimed_states = {
	input = game:all_states

	operators = {
		limit = {
			OR = {
				is_core_of = ROOT
				is_claimed_by = ROOT
				set_temp_variable = {
					tested_state = THIS
				}
				ROOT = {
					any_allied_country = {
						var:tested_state = {
							OR = {
								is_core_of = PREV
								is_claimed_by = PREV
							}
						}
					}
				}
			}
		}
	}

	name = COLLECTION_faction_allies_core_and_claimed_states
}
