faction_owned_core_states = {
	input = game:scope
	operators = {
		faction_members
		owned_states
		limit = {
			is_core_of = PREV 
		}
	}
	name = COLLECTION_FACTION_OWNED_CORE_STATES
}

faction_allies_core_and_claimed_states = {
	input = game:all_states

	operators = {
		limit = {
			OR = {
				is_core_of = ROOT
				is_claimed_by = ROOT
				set_temp_variable = {
					tested_state = THIS
				}
				ROOT = {
					any_allied_country = {
						var:tested_state = {
							OR = {
								is_core_of = PREV
								is_claimed_by = PREV
							}
						}
					}
				}
			}
		}
	}

	name = COLLECTION_faction_allies_core_and_claimed_states
}

faction_allies_controlled_core_and_claimed_states = {
	input = collection:faction_allies_core_and_claimed_states
	operators = {
		limit = {
			is_controlled_by_ROOT_or_ally = yes
		}
	}
	name = COLLECTION_faction_allies_controlled_core_and_claimed_states
}
