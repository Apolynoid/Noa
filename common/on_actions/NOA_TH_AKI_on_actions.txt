on_actions = {
	on_war = {
		effect = {
			if = {
				limit = {
					original_tag = AKI
					has_dynamic_modifier = {
						modifier = AKI_tu_di_jian_bing_idea
					}
				}
				AKI_remove_land_annexation_ic_debuff = yes
			}
		}
	}

	on_peace = {
		effect = {
			if = {
				limit = {
					original_tag = AKI
				}
				AKI_add_land_annexation_ic_debuff = yes
			}
		}
	}

	on_weekly_AKI = {
		effect = {
			set_variable = {
				AKI_army_defection_mtth = 0
			}
			# mtth = 350 * (1 - surrender_progress) ^ 2 + 10
			set_temp_variable = {
				t = 1.0
			}
			subtract_from_temp_variable = {
				t = AKI.surrender_progress
			}
			set_variable = {
				AKI_army_defection_mtth = 350
			}
			multiply_variable = {
				AKI_army_defection_mtth = t
			}
			multiply_variable = {
				AKI_army_defection_mtth = t
			}
			add_to_variable = {
				AKI_army_defection_mtth = 10
			}
		}
	}
}
