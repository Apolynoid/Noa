# netherworld civil war ai
KNA_civil_war_no_rush_without_buff = {
	allowed = {
		original_tag = YYK
	}
	
	enable = {
		has_country_flag = YYK_KNA_flag
		has_global_flag = YYK_netherworld_civil_war_begun_flag
		NOT = {
			has_global_flag = YYK_netherworld_civil_war_ended_flag
		}
		NOT = {
			has_idea = KNA_zui_zhong_gong_shi_idea
		}
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		
		tag = YKO

		ratio = 0.1

		priority = 10
		ordertype = front
		execution_type = careful
		execute_order = yes
	}
}

	abort_when_not_enabled = yes

	}

		type = diplo_action_acceptance
