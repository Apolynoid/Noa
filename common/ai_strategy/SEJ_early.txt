SEJ_do_not_join_war_in_SEJ_sphere = {
	allowed = {
		always = yes
	}
	
	enable = {
		# only for gekokujyo members
		is_in_faction_with = SEJ
		
		SEJ = {
			# SEJ at war with sphere members
			OR = {
				AND = {
					has_war_with = WHR
					WHR = {
						is_in_faction = no
					}
				}
				AND = {
					has_war_with = HKR
					HKR = {
						is_in_faction = no
					}
				}
			}
			# nothing special
			NOT = {
				surrender_progress > 0.1
			}
		}		
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = dont_join_wars_with
		id = SEJ
		value = 100
		target_country = WHR
	}

	ai_strategy = {
		type = dont_join_wars_with
		id = SEJ
		value = 100
		target_country = HKR
	}
}
