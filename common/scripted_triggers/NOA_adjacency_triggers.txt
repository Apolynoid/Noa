	OR = {
		has_tech = kongjianmonijishu
		any_allied_country = {
			has_tech = kongjianmonijishu
		}
		any_subject_country = {
			has_tech = kongjianmonijishu
		}
		any_country = {
			gives_military_access_to = ROOT
			OR = {
				any_allied_country = {
					has_tech = kongjianmonijishu
				}
				any_subject_country = {
					has_tech = kongjianmonijishu
				}
			}
		}
	}
}
