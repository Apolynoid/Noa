YYK_still_equipment_left_to_seize = {
	custom_trigger_tooltip = {
		tooltip = YYK_still_equipment_left_to_seize_tt
		set_temp_variable = {
			YYK_total_seized_equipment = KNA_equipment
		}
		add_to_temp_variable = {
			YYK_total_seized_equipment = YKO_secured_equipment
		}
		check_variable = {
			YYK_total_seized_equipment < 100
		}
	}
}

YYK_is_neighboring_core_state = {
	custom_trigger_tooltip = {
		tooltip = YYK_is_neighboring_core_state_tt
		OR = {
			AND = {
				ROOT = {
					has_country_flag = YYK_KNA_selected_flag
				}
				any_neighbor_state = {
					check_variable = {
						yko_control_level = 1
					}
				}
			}
			AND = {
				ROOT = {
					has_country_flag = YYK_YKO_selected_flag
				}
				any_neighbor_state = {
					check_variable = {
						yko_control_level = 4
					}
				}
			}
		}
	}
}
