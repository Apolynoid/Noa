defined_text = { 
	name = SRB_get_SCA_influence
	text = {
		trigger = {
			SCA = {
				OR = {
					NOA_is_independent = no
					has_country_flag = SRB_cancelled_investment_flag
				}
			}
		}
		localization_key = SRB_cancelled_investment_tt
	}
	text = {
		localization_key = "[?SCA.SRB_influence_value]"
	}
}
