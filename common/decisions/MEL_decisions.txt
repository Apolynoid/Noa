	# 特殊作物收获
	MEL_te_shu_zuo_wu_shou_huo = {
        allowed = {
            original_tag = MEL
        }
        activation = {
			always = no
		}
		available = {
			always = no
		}
		cancel_trigger = {
			always = no
		}

		icon = generic_plant
		is_good = yes
		days_mission_timeout = 120
		selectable_mission = no

		timeout_effect = {
			set_temp_variable = {
				MEL_toxic_crop_gain = 0
			}
			set_temp_variable = {
				MEL_medicinal_crop_gain = 0
			}
			hidden_effect = {
                if = {
                    limit = {
                        all_controlled_state = {
                            if = {
                                limit = {
                                    has_dynamic_modifier = {
                                        modifier = MEL_du_xing_zuo_wu_zhong_zhi_qu_idea
                                    }
                                }
                                add_to_temp_variable = {
                                    MEL_toxic_crop_gain = 1
                                }
                            }
                            if = {
                                limit = {
                                    has_dynamic_modifier = {
                                        modifier = MEL_yao_yong_zuo_wu_zhong_zhi_qu_idea
                                    }
                                }
                                add_to_temp_variable = {
                                    MEL_medicinal_crop_gain = 1
                                }
                            }
                            always = yes
                        }
                    }
                    log = "MEL crop gain [?MEL_toxic_crop_gain] [?MEL_medicinal_crop_gain]"
                }
				add_to_variable = {
					MEL_toxic_crop_num = MEL_toxic_crop_gain
				}
				add_to_variable = {
					MEL_medicinal_crop_num = MEL_medicinal_crop_gain
				}
			}
			custom_effect_tooltip = MEL_crop_gain_tt
			hidden_effect = {
				activate_mission = MEL_te_shu_zuo_wu_shou_huo
			}
		}
    }
